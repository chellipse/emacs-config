#+begin_src emacs-lisp
  (defmacro load! (file)
    "Load user-emacs-directory/FILE."
    `(load-file (concat (expand-file-name user-emacs-directory) ,file)))

  (defmacro after! (files &rest body)
    "Eval BODY after loading all FILES."
    (if (null files)
        `(progn ,@body)
      `(with-eval-after-load ',(car files)
         (after! ,(cdr files) ,@body)))
    )

  (defmacro add-hook! (files target)
    "Define hooks for all FILES to TARGET...
  FILES should be an unquoted list.
  TARGET should be a quoted mode"
    (cl-loop for val in files
             collect `(add-hook ',val ,target) into defs
             finally return `(progn ,@defs)))

  (defmacro cmd! (&rest body)
    "Returns (lambda () (interactive) ,@body)
  A factory for quickly producing interaction commands, particularly for keybinds
  or aliases."
    (declare (doc-string 1))
    `(lambda (&rest _) (interactive) ,@body))

  (defvar doom-evil-state-alist
    '((?n . normal)
      (?v . visual)
      (?i . insert)
      (?e . emacs)
      (?o . operator)
      (?m . motion)
      (?r . replace)
      (?g . global))
    "A list of cons cells that map a letter to a evil state symbol.")

  (defun doom-keyword-name (keyword)
    "Returns the string name of KEYWORD (`keywordp') minus the leading colon."
    (declare (pure t) (side-effect-free t))
    (cl-check-type keyword keyword)
    (substring (symbol-name keyword) 1))

  (defun doom--map-keyword-to-states (keyword)
    "Convert a KEYWORD into a list of evil state symbols.

  For example, :nvi will map to (list 'normal 'visual 'insert). See
  `doom-evil-state-alist' to customize this."
    (cl-loop for l across (doom-keyword-name keyword)
             if (assq l doom-evil-state-alist) collect (cdr it)
             else do (error "not a valid state: %s" l)))

  (defmacro my-general-evil-define-key (states keymaps prefix &rest args)
    "A wrapper for `general-define-key' that is similar to `evil-define-key'.
  It has positional arguments for STATES and KEYMAPS (that will not be overridden
  by a later :keymaps or :states argument). Besides this, it acts the same as
  `general-define-key', and ARGS can contain keyword arguments in addition to
  keybindings. This can basically act as a drop-in replacement for
  `evil-define-key', and unlike with `general-define-key', KEYMAPS does not need
  to be quoted."
    (declare (indent 2))
    `(general-def
       :states ,(if (and (listp states)
                         (eq (car states) 'quote))
                    `,states
  		        `',states)
       :keymaps ,(if (and (listp keymaps)
  			            (eq (car keymaps) 'quote))
                     `,keymaps
                   `',keymaps)
       :prefix ,(if (and (listp prefix)
                         (eq (car prefix) 'quote))
                    `,prefix
  		        `',prefix)
       ,@args))

  (defmacro map! (states mode &rest body)
    "A wrapper around my-general-evil-define-key."
    `(my-general-evil-define-key ,(doom--map-keyword-to-states states) ,mode () ,@body))

  (defmacro map-leader! (states mode leader &rest body)
    "A wrapper around my-general-evil-define-key."
    `(my-general-evil-define-key ,(doom--map-keyword-to-states states) ,mode ,leader ,@body))

  (defmacro map-after! (files states mode &rest body)
    "A wrapper around my-general-evil-define-key."
    `(after! ,files (my-general-evil-define-key ,(doom--map-keyword-to-states states) ,mode () ,@body)))

  (defmacro map-leader-after! (files states mode leader &rest body)
    "A wrapper around my-general-evil-define-key."
    `(after! ,files (my-general-evil-define-key ,(doom--map-keyword-to-states states) ,mode ,leader ,@body)))

#+begin_src emacs-lisp
  (defun reload-config ()
    "Reload the current config's init.el."
    (interactive)
    (load-file (expand-file-name "init.el" user-emacs-directory)))

  ;; All defined under UI > Font, which tangles after this block; declared
  ;; here so the byte-compiler knows the font functions below are touching
  ;; dynamic variables rather than typos.
  (defvar font-size)
  (defvar default-font-mono)
  (defvar default-font-propo)
  (defvar my/font-mono-candidates)
  (defvar my/font-propo-candidates)
  (defvar my/font-cjk-candidates)

  (defvar my/font-families-cache nil
    "Memoized result of `my/font-families'.")

  (defun my/font-families (&optional refresh)
    "Return every installed font family, according to fc-list.
Used in preference to `font-family-list' because that one needs a
graphical display, and the daemon loads this config before any display
exists.  fc-list also reports width variants that `font-family-list'
hides -- see `select-global-font'.  With REFRESH, re-run fc-list rather
than reusing the cached answer."
    (when (or refresh (null my/font-families-cache))
      (setq my/font-families-cache
            (ignore-errors
              (delete-dups
               (mapcar #'string-trim
                       (seq-mapcat (lambda (line) (split-string line ","))
                                   (process-lines "fc-list" ":" "family")))))))
    my/font-families-cache)

  (defun my/font-available-p (family)
    "Return non-nil if FAMILY is installed.
Asks the font backend when there is a graphical frame to ask, and
fc-list otherwise: without a display every font query answers nil, so
the backend would report each and every family as missing."
    (and (stringp family)
         (if (display-graphic-p)
             (and (find-font (font-spec :family family)) t)
           (and (member family (my/font-families)) t))))

  (defun my/resolve-font (candidates)
    "Return the first installed family in CANDIDATES.
When none of them is installed, returns the last one, which is a
fontconfig generic alias -- so callers always get a usable string."
    (or (seq-find #'my/font-available-p candidates)
        (car (last candidates))))

  (defun refresh-global-font-size ()
    (interactive)
    ;; Construct a fontconfig pattern: "FontName-Size[:attr=val...]".
    ;; The suffix (see `my/font-mono-pattern-attrs') lets a family pick a
    ;; non-default weight or width while faces still inherit the family
    ;; itself, so bold/italic faces resolve to that family's real cuts.
    (let ((font-string (format "%s-%d%s" default-font-mono font-size
                               (or (cdr (assoc default-font-mono
                                               my/font-mono-pattern-attrs))
                                   ""))))
      ;; Update all existing frames.  Guarded because `set-frame-font'
      ;; signals "Font not available" for a family that isn't installed,
      ;; and uncaught during startup that aborts the rest of init.  Skip
      ;; terminal frames, which have no font to set.
      (when (condition-case err
                (progn (dolist (frame (frame-list))
                         (when (display-graphic-p frame)
                           (set-frame-font font-string nil (list frame))))
                       t)
              (error (message "Could not set font %s: %s"
                              font-string (error-message-string err))
                     nil))
        ;; Update default for new frames -- but only with a font that just
        ;; applied cleanly, since `make-frame' signals on one it cannot
        ;; open and would then fail for every new frame.  `setf' rather
        ;; than `add-to-list' so this replaces the previous font instead of
        ;; leaving a stale duplicate `font' key behind, which `make-frame'
        ;; would still try to honour.
        (setf (alist-get 'font default-frame-alist) font-string))
      ;; Force redisplay to fix line numbers and other rendering
      (redraw-display)))

  (defun increase-global-font-size ()
    (interactive)
    (setq font-size (+ font-size 1))
    (refresh-global-font-size)
    (message "Font size: %d" font-size))

  (defun decrease-global-font-size ()
    (interactive)
    (setq font-size (- font-size 1))
    (refresh-global-font-size)
    (message "Font size: %d" font-size))

  (defun select-global-font (&optional proportional)
    "Pick a system font via completion and apply it globally.
Completes over every installed font family (orderless matching
applies via `completion-styles').  Families are enumerated with
fc-list rather than `font-family-list', because the latter only
reports each face's first (typographic) family name and so hides
width variants like \"Ioskeley Mono Condensed\".  With a prefix
argument set the proportional/`variable-pitch' font instead of
the monospace font."
    (interactive "P")
    (let* ((families (seq-sort #'string< (my/font-families t)))
           (font (completing-read
                  (format "%s font: " (if proportional "Proportional" "Monospace"))
                  families nil t)))
      (if proportional
          (progn
            (setq default-font-propo font)
            (set-face-attribute 'variable-pitch t :family font)
            (message "Proportional font: %s" font))
        (setq default-font-mono font)
        (refresh-global-font-size)
        (message "Monospace font: %s" font))))

  (defun my/apply-fonts (&rest _)
    "Point the font variables at installed families, and apply them everywhere."
    (setq default-font-mono (my/resolve-font my/font-mono-candidates)
          default-font-propo (my/resolve-font my/font-propo-candidates))
    (refresh-global-font-size)
    (set-face-attribute 'variable-pitch t :family default-font-propo)
    ;; Use a scalable CJK font instead of falling back to bitmap Unifont
    (let ((cjk (my/resolve-font my/font-cjk-candidates)))
      (dolist (charset '(han kana hangul cjk-misc bopomofo))
        (set-fontset-font t charset (font-spec :family cjk)))))

  (defun my/apply-fonts-once ()
    "Re-apply fonts on the first graphical frame, then stop listening."
    (when (display-graphic-p)
      (remove-hook 'server-after-make-frame-hook #'my/apply-fonts-once)
      (my/apply-fonts)))

  (defun search-wikipedia ()
    (interactive)
    (let ((search (read-string "Search: ")))
      (w3m (concat "https://en.wikipedia.org/wiki/Special:Search?search=" search))))

  (defun copy-file-path ()
    "Copy the current buffer's file path and line number to the clipboard.
The path is abbreviated with `~' when it lives under the home directory,
and the result is formatted as PATH:LINE so it can be pasted straight into
a shell, a compiler error list, or an agent prompt."
    (interactive)
    (if-let* ((file (buffer-file-name)))
        ;; `line-number-at-pos' with ABSOLUTE ignores narrowing, so a
        ;; narrowed buffer still reports the real line in the file.
        (let ((path (format "%s:%d"
                            (abbreviate-file-name file)
                            (line-number-at-pos nil t))))
          (kill-new path)
          (message "Copied: %s" path))
      (message "Buffer is not visiting a file")))

  (defun ediff-current-buffer ()
    "Run ediff-buffers with current buffer as first argument."
    (interactive)
    (ediff-buffers (current-buffer)
                   (read-buffer "Buffer to compare with: " nil t)))

#+begin_src emacs-lisp
  (defvar journal-file-path "~/journal.org.age"
    "Path to the journal file.")

  (defun new-journal-entry ()
    "Open the journal file and start a new timestamped entry."
    (interactive)
    (find-file journal-file-path)
    (goto-char (point-max))
    ;; Trim trailing whitespace at end of buffer
    (while (and (not (bobp)) (memq (char-before) '(?\n ?\s ?\t)))
      (delete-char -1))
    ;; Two blank lines to separate from previous content
    (insert "\n* ")
    ;; Date header
    (insert (string-trim-right
             (shell-command-to-string "date '+%F(%a) %H·%M%:::z'")))
    (insert "\n")
    ;; Mark where we want the cursor
    (let ((entry-point (point)))
      ;; Add 3 newlines for breathing room
      (insert "\n\n\n")
      (goto-char entry-point)))

** Journal Refill
#+begin_src emacs-lisp
  ;; Live-reflow the *current* org paragraph as you edit -- an org-aware
  ;; alternative to built-in `refill-mode', which only reliably refills the
  ;; paragraph you're appending to (its overlay/tail optimisation and
  ;; space-gated triggering break when you go back to edit an earlier one).
  ;; This just runs `org-fill-paragraph' on whatever paragraph point is in,
  ;; after every editing command.
  (defvar-local journal-refill--doit nil
    "Set by after-change; consumed by post-command to trigger a refill.")
  (defvar-local journal-refill--filling nil
    "Non-nil while we are the ones filling, to avoid re-triggering.")

  (defun journal-refill--safe-p ()
    "Non-nil when point sits in an ordinary paragraph safe to auto-fill."
    (and (not (org-at-table-p))
         (not (org-at-heading-p))
         (not (org-in-src-block-p))
         (not (org-at-comment-p))
         (not (org-at-keyword-p))
         (not (org-at-block-p))
         ;; Don't fill while point follows whitespace -- otherwise the space
         ;; you just typed to end a word gets collapsed and you can't start
         ;; the next word. Filling resumes on the next real character.
         (not (memq (char-before) '(?\s ?\t)))
         (memq (org-element-type (org-element-at-point))
               '(paragraph item plain-list))))

  (defun journal-refill--post-command ()
    "Reflow the paragraph at point if an edit just happened and it's safe."
    (when (and journal-refill--doit (not journal-refill--filling))
      (setq journal-refill--doit nil)
      (ignore-errors
        (when (and journal-refill-mode
                   (not buffer-read-only)
                   (not (region-active-p))
                   (derived-mode-p 'org-mode)
                   (journal-refill--safe-p))
          (let ((journal-refill--filling t))
            (save-excursion (org-fill-paragraph)))))))

  (defun journal-refill--after-change (&rest _)
    "Flag that a real edit happened (skipping our own fills and undo)."
    (unless (or journal-refill--filling
                (memq this-command '(undo undo-redo undo-tree-undo
                                          undo-tree-redo evil-undo evil-redo)))
      (setq journal-refill--doit t)))

  (define-minor-mode journal-refill-mode
    "Continuously reflow the current org paragraph as you edit, like a live `gqap'.
  Uses `org-fill-paragraph', and skips tables, source blocks, headings,
  keywords, and other structured elements."
    :lighter " ↩fill"
    (if journal-refill-mode
        (progn
          (add-hook 'after-change-functions #'journal-refill--after-change nil t)
          (add-hook 'post-command-hook #'journal-refill--post-command nil t))
      (remove-hook 'after-change-functions #'journal-refill--after-change t)
      (remove-hook 'post-command-hook #'journal-refill--post-command t)
      (setq journal-refill--doit nil)))

  (defun journal-refill--maybe-enable ()
    "Enable `journal-refill-mode' only in the journal file."
    (when (and buffer-file-name
               (ignore-errors (file-equal-p buffer-file-name journal-file-path)))
      (journal-refill-mode 1)))
  (add-hook 'org-mode-hook #'journal-refill--maybe-enable)
#+end_src

** History
#+begin_src emacs-lisp
  ;; Persist EWW's visited-URL history across restarts. `eww' builds its
  ;; prompt completion table from `eww-prompt-history' (a plain defvar that
  ;; normally resets each session), so we (a) record every rendered page --
  ;; not just URLs typed at the prompt, but links clicked through too -- and
  ;; (b) have savehist save the variable to disk.
  (with-eval-after-load 'eww
    (defun my/eww-record-url ()
      "Add the current page's URL to `eww-prompt-history' for completion."
      (when-let* ((url (plist-get eww-data :url)))
        (add-to-history 'eww-prompt-history url 1000)))
    (add-hook 'eww-after-render-hook #'my/eww-record-url)

    (defun my/eww-forget-url (url)
      "Permanently remove URL from `eww-prompt-history'.
  Completes over history (defaulting to the current page) and writes the
  savehist file immediately, so the entry won't come back on restart --
  though visiting the page again re-records it."
      (interactive
       (list (completing-read "Forget EWW URL: " eww-prompt-history nil t nil nil
                              (and (derived-mode-p 'eww-mode)
                                   (plist-get eww-data :url)))))
      (setq eww-prompt-history (delete url eww-prompt-history))
      (when (bound-and-true-p savehist-mode) (savehist-save))
      (message "Forgot %s" url)))

  ;; Address-bar behaviour for the EWW URL prompt: preselect what you TYPED,
  ;; not a longer visited URL that has it as a prefix -- so RET visits
  ;; "example.com/" even while "example.com/writing.html" is in history.
  ;; Suggestions are still reachable with C-n / down. Keyed on the prompt's
  ;; history variable so it only affects `eww', not other minibuffers.
  ;; (M-RET / `vertico-exit-input' does the same ad hoc in any minibuffer.)
  ;;
  ;; Preserve visit order in the shared EWW/w3m history. Prescient tracks
  ;; prompt selections only, omitting followed links and newly typed URLs;
  ;; disable its sorting and tracking for this prompt.
  (defun my/eww-prompt-preselect-input ()
    "Preselect typed input and preserve visit order in EWW/w3m URL prompts."
    (when (eq minibuffer-history-variable 'eww-prompt-history)
      (setq-local vertico-preselect 'prompt
                  vertico-sort-function nil)
      (remove-hook 'post-command-hook
                   #'vertico-prescient--remember-inserted-candidate t)
      (remove-hook 'minibuffer-exit-hook
                   #'vertico-prescient--remember-exited-candidate t)))
  (add-hook 'minibuffer-setup-hook #'my/eww-prompt-preselect-input)

  (savehist-mode 1)
  (add-to-list 'savehist-additional-variables 'eww-prompt-history)
  ;; per-project last compile commands, defined under Langs > prog-mode.
  (add-to-list 'savehist-additional-variables 'my-compile-last-commands)

#+begin_src emacs-lisp
  ;; Each list is tried in order and the first family actually installed
  ;; wins, so a font going missing -- a rebuild that drops a font package,
  ;; say -- falls through to the next choice instead of erroring out of
  ;; init.el and leaving the rest of this config unloaded.  Every list ends
  ;; in a fontconfig generic alias, which always resolves to something.
  (defvar my/font-mono-candidates
    '("Berkeley Mono"
      "Ioskeley Mono Semi-Condensed"
      "Iosevka"
      "DejaVu Sans Mono"
      "monospace")
    "Monospace families to try, most preferred first.")

  ;; USGC flattens every weight/width cut of Berkeley Mono into its own
  ;; single-face family ("Berkeley Mono Medium SemiCondensed", ...), and a
  ;; single-face family can never satisfy a bold or italic request with a
  ;; real font -- fontconfig answers with synthetic embolden/shear of the
  ;; one face instead.  So use the umbrella family above and express the
  ;; preferred cut as pattern attributes here: faces that want bold or
  ;; oblique then resolve to the genuine cuts along those axes.
  (defvar my/font-mono-pattern-attrs
    '(("Berkeley Mono" . ":weight=medium:width=semicondensed"))
    "Fontconfig pattern suffix per monospace family, applied when that
family is the one `refresh-global-font-size' sets.")

  (defvar my/font-propo-candidates
    '("Roboto"
      "Noto Sans"
      "DejaVu Sans"
      "sans-serif")
    "Proportional families to try, most preferred first.")

  (defvar my/font-cjk-candidates
    '("Noto Sans Mono CJK SC"
      "Noto Sans CJK SC"
      "monospace")
    "CJK families to try, most preferred first.")

  (defvar default-font-mono nil
    "Monospace family in use, chosen by `my/apply-fonts'.")
  (defvar default-font-propo nil
    "Proportional family in use, chosen by `my/apply-fonts'.")

  (setq default-font-size 12)
  (unless (boundp 'font-size)
    (setq font-size default-font-size))

  ;; Emacs simulates missing bold cuts with a fixed one-pixel overstrike
  ;; (`overstrike' in xfaces.c), which smears glyphs. For monospace families
  ;; with no cut heavier than medium, override bold faces with `:weight normal'.
  ;; Overrides take precedence over theme and defface specs without changing
  ;; other attributes. Restore previous overrides when the family changes.
  (defvar my/fake-bold-saved nil
    "Alist of (FACE . PREVIOUS-OVERRIDE-SPEC) for faces made non-bold.")

  (defvar my/fake-bold-family nil
    "Last family processed by `my/font-suppress-fake-bold'.")

  (defun my/font-has-bold-p (family)
    "Return non-nil if FAMILY has a cut heavier than medium (fontconfig weight 100).
  Treat unlisted families, including generic aliases, as having bold cuts."
    (let ((lines (ignore-errors
                   (process-lines "fc-list"
                                  ;; Escape fontconfig pattern syntax; - starts a size.
                                  (concat ":family="
                                          (replace-regexp-in-string
                                           "[-:,\\]" "\\\\\\&" family))
                                  "weight"))))
      (or (null lines)
          (seq-some (lambda (line)
                      (seq-some (lambda (n) (> (string-to-number n) 100))
                                (split-string line "[^0-9]+" t)))
                    lines))))

  (defun my/fake-bold--override (spec)
    "Return override SPEC with `:weight normal' added.
  Return nil for unsupported display clauses; nil SPEC gets a default clause."
    (pcase spec
      ('nil '((t :weight normal)))
      (`((t . ,attrs))
       ;; Old-style clauses wrap attributes in a single plist.
       (when (and (consp (car attrs)) (null (cdr attrs)))
         (setq attrs (car attrs)))
       `((t ,@attrs :weight normal)))))

  (defun my/font-suppress-fake-bold (&optional force)
    "Make bold faces normal weight when `default-font-mono' has no bold cut.
  Skip unchanged families unless FORCE is non-nil."
    (when (or force (not (equal default-font-mono my/fake-bold-family)))
      (setq my/fake-bold-family default-font-mono)
      ;; Restore overrides before checking face weights.
      (pcase-dolist (`(,face . ,spec) my/fake-bold-saved)
        (face-spec-set face spec 'face-override-spec))
      (setq my/fake-bold-saved nil)
      (unless (my/font-has-bold-p default-font-mono)
        (dolist (face (face-list))
          (when (memq (face-attribute face :weight nil)
                      '(semi-bold demibold bold extra-bold extrabold
                        ultra-bold ultrabold heavy black))
            (let* ((saved (get face 'face-override-spec))
                   (spec (my/fake-bold--override saved)))
              (when spec
                (push (cons face saved) my/fake-bold-saved)
                (face-spec-set face spec 'face-override-spec))))))))
  (advice-add 'refresh-global-font-size :after
              (lambda (&rest _) (my/font-suppress-fake-bold))
              '((name . my/fake-bold)))
  ;; Recheck face weights after theme changes.
  (add-hook 'enable-theme-functions
            (lambda (theme)
              (unless (eq theme 'user)
                (my/font-suppress-fake-bold t))))

  (my/apply-fonts)

  ;; The daemon gets this far with no display, so the families above were
  ;; resolved against fc-list rather than the font backend.  Only the
  ;; backend's opinion decides whether `make-frame' succeeds, and a font it

** Merge disk changes
Auto-revert skips modified buffers. =my/merge-disk-changes= merges unsaved edits
with disk changes, reconstructing the base from undo history instead of storing
a snapshot. The =(t . MODTIME)= entry identifies the last synchronized state;
undo-tree stores it in a node's changeset.

Apply the disk version as an unmodified state, then the merged edits as a
separate undo step. One undo exposes the disk version and preserves a base for
subsequent merges. Conflicts use diff3 markers and =smerge-mode=.
#+begin_src emacs-lisp
  (defun my/merge--sync-marker-p (entry)
    "Non-nil if undo ENTRY marks the state matching the visited file's modtime."
    (and (eq (car-safe entry) t)
         (ignore-errors (time-equal-p (cdr entry) (visited-file-modtime)))))

  (defun my/merge--edits-since-sync ()
    "Return undo entries restoring this buffer's last synchronized disk state.
  Signal a `user-error' if undo history no longer reaches that state."
    (let ((entries nil) (found nil))
      (if (bound-and-true-p undo-tree-mode)
          (progn
            (undo-list-transfer-to-tree)
            ;; Each node's changeset undoes the transition from its parent.
            (let ((node (undo-tree-current buffer-undo-tree)))
              (while (and node (not found))
                (let ((changeset (undo-tree-node-undo node)))
                  (setq entries (append entries (copy-sequence changeset))
                        found (seq-some #'my/merge--sync-marker-p changeset)
                        node (undo-tree-node-previous node))))))
        (let ((list buffer-undo-list))
          (while (and (consp list) (not found))
            (push (car list) entries)
            (setq found (my/merge--sync-marker-p (car list))
                  list (cdr list)))
          (setq entries (nreverse entries))))
      (unless found
        (user-error "Undo history no longer reaches the version last read from disk"))
      ;; Drop boundaries that would stop `primitive-undo' early and marker
      ;; adjustments that would move the source buffer's markers. Undo-tree
      ;; represents these markers as (undo-tree-idN . ADJUSTMENT); neither
      ;; form affects text.
      (seq-remove (lambda (e)
                    (or (null e)
                        (and (consp e)
                             (or (markerp (car e))
                                 (and (car e) (symbolp (car e)) (not (eq (car e) t))
                                      (numberp (cdr e)))))))
                  entries)))

  (defun my/merge--dedupe-sync-markers ()
    "Keep only the oldest unmodified marker in the newest undo group.
  `replace-buffer-contents' can add a mid-replacement marker whose undo
  creates a redo entry that incorrectly marks the buffer unmodified."
    (let* ((group (seq-take-while #'identity buffer-undo-list))
           (markers (seq-filter (lambda (e) (eq (car-safe e) t)) group)))
      (when (cdr markers)
        (let ((keep (car (last markers))))
          (setq buffer-undo-list
                (append (seq-remove (lambda (e) (and (eq (car-safe e) t)
                                                     (not (eq e keep))))
                                    group)
                        (nthcdr (length group) buffer-undo-list)))))))

  (defun my/merge--write (buffer file)
    "Write BUFFER's text to FILE losslessly, ending it with a newline.
Return non-nil if the text already ended with one (or is empty)."
    (with-current-buffer buffer
      (save-restriction
        (widen)
        (let ((coding-system-for-write 'utf-8-emacs-unix)
              (newline (or (= (point-min) (point-max))
                           (eq (char-before (point-max)) ?\n))))
          (write-region nil nil file nil 'silent)
          ;; diff3 requires a final newline to separate text from conflict markers.
          (unless newline
            (write-region "\n" nil file t 'silent))
          newline))))

  (defun my/merge-disk-changes ()
    "Three-way merge this buffer's unsaved edits with the file's disk version.
  Reconstruct the base from undo history. Conflicts use diff3 markers and
  enable `smerge-mode'."
    (interactive)
    (unless buffer-file-name
      (user-error "Buffer isn't visiting a file"))
    (when (my/encrypted-file-buffer-p)
      (user-error "Won't merge an encrypted file: diff3 needs plaintext copies"))
    (unless (file-exists-p buffer-file-name)
      (user-error "File no longer exists on disk"))
    (cond
     ((verify-visited-file-modtime)
      (message "File is unchanged on disk; nothing to merge"))
     ((not (buffer-modified-p))
      (revert-buffer t t t)
      (message "No unsaved edits; reverted to the disk version"))
     (t
      (save-restriction
        (widen)
        (let* ((edits (my/merge--edits-since-sync))
               (disk-time (file-attribute-modification-time
                           (file-attributes buffer-file-name)))
               (dir (or (getenv "XDG_RUNTIME_DIR") temporary-file-directory))
               (mine (make-temp-file (expand-file-name "merge-mine-" dir)))
               (base (make-temp-file (expand-file-name "merge-base-" dir)))
               (theirs (make-temp-file (expand-file-name "merge-disk-" dir)))
               (base-buf (generate-new-buffer " *merge base*" t))
               (disk-buf (generate-new-buffer " *merge disk*" t))
               (merged-buf (generate-new-buffer " *merge result*" t))
               (orig (current-buffer))
               nl-mine nl-base nl-disk status conflicts)
          (unwind-protect
              (progn
                ;; Reconstruct in a scratch buffer to preserve source text and history.
                (with-current-buffer base-buf
                  (insert-buffer-substring-no-properties orig)
                  (condition-case err
                      (primitive-undo 1 edits)
                    (error (user-error "Couldn't rebuild the disk version from undo: %s"
                                       (error-message-string err)))))
                (with-current-buffer disk-buf
                  (let ((coding-system-for-read
                         (buffer-local-value 'buffer-file-coding-system orig)))
                    (insert-file-contents (buffer-local-value 'buffer-file-name orig))))
                (setq nl-mine (my/merge--write orig mine)
                      nl-base (my/merge--write base-buf base)
                      nl-disk (my/merge--write disk-buf theirs))
                (with-current-buffer merged-buf
                  (let ((coding-system-for-read 'utf-8-emacs-unix))
                    (setq status (call-process "diff3" nil (list t nil) nil "-m"
                                               "-L" "buffer" "-L" "base" "-L" "disk"
                                               mine base theirs)))
                  ;; Use the disk's final-newline state unless the buffer changed
                  ;; it from the base. Preserve diff3's newline on conflicts.
                  (when (and (eq status 0)
                             (not (if (eq nl-mine nl-base) nl-disk nl-mine))
                             (eq (char-before (point-max)) ?\n))
                    (goto-char (point-max))
                    (delete-char -1)))
                (unless (memq status '(0 1))
                  (user-error "diff3 failed (exit %s)" status))
                ;; Record the disk version as the synchronized undo state.
                (replace-buffer-contents disk-buf 1.0)
                (set-visited-file-modtime disk-time)
                (set-buffer-modified-p nil)
                (undo-boundary)
                ;; Keep the merged edits in a separate undo step.
                (replace-buffer-contents merged-buf 1.0)
                (setq conflicts (= status 1)))
            (mapc #'delete-file (list mine base theirs))
            (mapc #'kill-buffer (list base-buf disk-buf merged-buf)))
          (if (not conflicts)
              (message "Merged disk changes%s"
                       (if (buffer-modified-p) " under your edits" "; no edits of yours remain"))
            (smerge-mode 1)

** Encrypted-file hardening
undo history and org element caches can persist decrypted buffer data under
var/ without using encryption handlers. detect encrypted buffers through
file-name-handler-alist rather than extensions, and disable these disk caches
for files handled by age or easypg. see the undo-tree and org hooks below.
#+begin_src emacs-lisp
  (defun my/encrypted-file-buffer-p ()
    "return non-nil if the buffer uses an age or easypg write handler.
detect encryption through file-name-handler-alist rather than extensions."
    (and buffer-file-name
         (memq (find-file-name-handler buffer-file-name 'write-region)
               '(age-file-handler epa-file-handler))))

  ;; org-persist writes headline text and structure under var/org/persist/.
  ;; disable persistent caching for encrypted buffers, then reset the cache to
  ;; unregister existing disk data. the in-memory cache remains enabled.
  (with-eval-after-load 'org
    (add-hook 'org-mode-hook
              (lambda ()
                (when (my/encrypted-file-buffer-p)
                  (setq-local org-element-cache-persistent nil)
                  (when (fboundp 'org-element-cache-reset)
                    (org-element-cache-reset))))))

  ;; audit possible plaintext derivatives under var/. undo, backup and autosave
  ;; filenames often encode the source path. inspect only names with encrypted
  ;; extensions, then check for ciphertext headers. this is a heuristic, not a
  ;; complete leak scan.
  (defun my/file-encrypted-on-disk-p (file)
    "return non-nil if the file header resembles age or openpgp ciphertext."
    (ignore-errors
      (with-temp-buffer
        (let ((coding-system-for-read 'no-conversion))
          (insert-file-contents-literally file nil 0 64))
        (and (> (buffer-size) 0)
             (or (save-excursion (search-forward "age-encryption.org" nil t))
                 (save-excursion (search-forward "BEGIN AGE ENCRYPTED" nil t))
                 (save-excursion (search-forward "BEGIN PGP MESSAGE" nil t))
                 ;; binary openpgp packet tags have the high bit set.
                 (>= (char-after (point-min)) #x80))))))

  (defun my/audit-encrypted-leaks (&optional interactive)
    "warn about possible plaintext derivatives of encrypted files under var/.
return suspect paths using the my/file-encrypted-on-disk-p heuristic."
    (interactive "p")
    (let* ((root (ignore-errors (no-littering-expand-var-file-name "")))
           (enc-re "\\.\\(age\\|gpg\\|asc\\|pgp\\)\\(~\\|\\.~undo-tree~\\)?\\'")
           (leaks nil))
      (when (and root (file-directory-p root))
        (dolist (f (directory-files-recursively root ".*"))
          (let ((decoded (replace-regexp-in-string "!" "/" (file-name-nondirectory f))))
            (when (and (string-match-p enc-re decoded)
                       (not (my/file-encrypted-on-disk-p f)))
              (push f leaks)))))
      (dolist (f leaks)
        (display-warning
         'security
         (format "Plaintext derivative of an encrypted file on disk: %s" f)
         :error))
      (when interactive
        (message (if leaks "%d encrypted-file leak(s) found -- see *Warnings*"
                   "No plaintext leaks of encrypted files under var/")
                 (length leaks)))
      leaks))

  ;; run once on an idle timer to avoid blocking init or adding timers on
  ;; reload.
  (unless after-init-time
    (run-with-idle-timer 5 nil #'my/audit-encrypted-leaks))
#+end_src
** =undo-tree=
#+begin_src emacs-lisp
  (use-package undo-tree
    :demand t
    :after no-littering
    :config
    ;; Exclude externally generated buffers. Unlike
    ;; `undo-tree-incompatible-major-modes', include derived modes so comint
    ;; covers shell/agent-shell and magit-mode covers all Magit buffers.
    (defvar my/undo-tree-excluded-modes
      '(comint-mode eshell-mode vterm-mode term-mode magit-mode
        w3m-mode eww-mode special-mode)
      "Major modes excluded from `undo-tree-mode', including derived modes.")
    (define-advice turn-on-undo-tree-mode (:before-until (&rest _) my/excluded)
      (apply #'derived-mode-p my/undo-tree-excluded-modes))

    ;; Evil's u/C-r call undo-tree directly and reject buffers where it is
    ;; disabled. Bypass that check and use built-in undo/redo in those buffers.
    (define-advice undo-tree-undo (:around (orig &optional arg) my/fallback)
      (if undo-tree-mode (funcall orig arg) (undo-only arg)))
    (define-advice undo-tree-redo (:around (orig &optional arg) my/fallback)
      (if undo-tree-mode (funcall orig arg) (undo-redo arg)))
    (define-advice evil--check-undo-system (:before-until () my/fallback)
      (not (bound-and-true-p undo-tree-mode)))

    (global-undo-tree-mode 1)
    (setq undo-tree-auto-save-history t
          undo-tree-history-directory-alist
          `(("." . ,(no-littering-expand-var-file-name "undo-tree-hist/"))))
    ;; Don't warn about undo history conflicts between instances
    (setq undo-tree-enable-undo-in-region nil)

    ;; undo-tree defaults to 80, 120 and 360 mib limits, roughly 500 times the
    ;; emacs defaults. saving serializes the entire tree through
    ;; write-file-functions. external edits with auto-revert can add a
    ;; whole-buffer changeset per revert; config.org reached 610 mb of history
    ;; and 8.5 seconds per save. lower the limits to reduce history writes.
    (setq undo-tree-limit        (* 8 1024 1024)
          undo-tree-strong-limit (* 12 1024 1024)
          undo-tree-outer-limit  (* 24 1024 1024))

    ;; Org source fontification records text property changes
    ;; (nil PROP VAL BEG . END) containing syntax tables. undo-tree serializes
    ;; the shared table per entry, producing 2.7 GB of history for config.org.
    ;; Property changes do not shift positions; omit them from saved history
    ;; without affecting text undo. In-session undo retains them.
    (defun my/undo-tree-drop-property-changes (element)
      "Return nil for text property change ELEMENT, else ELEMENT."
      (unless (and (consp element) (null (car element)))
        element))
    (add-hook 'undo-tree-pre-save-element-functions
              #'my/undo-tree-drop-property-changes)

    ;; undo-tree writes inserted and deleted text under var/undo-tree-hist/
    ;; without encryption handlers. block history saves and loads for encrypted
    ;; buffers. checking the handler in both functions avoids dependence on hook
    ;; order, file extensions or per-buffer settings.
    (define-advice undo-tree-save-history
        (:before-while (&rest _) skip-encrypted-files)
      "skip persistent undo writes for encrypted buffers."
      (not (my/encrypted-file-buffer-p)))

    (define-advice undo-tree-load-history
        (:before-while (&rest _) skip-encrypted-files)
      "skip persistent undo reads for encrypted buffers, including old history."
      (not (my/encrypted-file-buffer-p))))

** =autoconf-mode=
#+begin_src emacs-lisp
  ;; Use autoconf-mode for .m4 macros (aclocal, Lustre's config/). m4-mode
  ;; pairs backticks and apostrophes as brackets, breaking paren matching
  ;; on shell strings and prose. autoconf-mode pairs only [] and ().
  (add-to-list 'auto-mode-alist '("\\.m4\\'" . autoconf-mode))

  ;; Treat unmatched ) in shell `case' arms (`yes)', `*)', `linux*)') as
  ;; punctuation while preserving autoconf-mode's `dnl' handling. Match only
  ;; single space-free patterns to exclude wrapped C argument lists in tests.
  (defconst my/autoconf-case-arm-re
    "^[ \t]*[^][()#,;\"' \t\n]+\\()\\)\\(?:[ \t]\\|$\\)"
    "A shell `case' arm; group 1 is its closing paren.")

  (defun my/autoconf-case-arms ()
    (let ((orig syntax-propertize-function))
      (setq-local syntax-propertize-function
                  (lambda (start end)
                    (when orig (funcall orig start end))
                    (goto-char start)
                    (while (re-search-forward my/autoconf-case-arm-re end t)
                      (put-text-property (match-beginning 1) (match-end 1)
                                         'syntax-table (string-to-syntax ".")))))))
  (add-hook 'autoconf-mode-hook #'my/autoconf-case-arms)
#+end_src
** =c/c++-ts-mode=
#+begin_src emacs-lisp
  ;; use the tree-sitter modes wherever emacs would pick cc-mode. c-or-c++
  ;; covers .h files, guessing the language from the contents.
  (dolist (remap '((c-mode . c-ts-mode)
                   (c++-mode . c++-ts-mode)

  ;; hooks that into every RET. clangd formats with clang-format, which
  ;; ignores the tabs preference lsp-mode sends and, in a tree without a
  ;; .clang-format (lustre), falls back to LLVM style: so after cc-mode has
  ;; already indented a new line correctly, the async reply rewrites it (and

  ;; same formatter. lsp reads both when it configures the buffer, which
  ;; happens on an idle timer after the mode hooks, so buffer-local values

  (add-hook! (c-mode-hook c-ts-mode-hook)
             (lambda ()
               (setq-local lsp-enable-on-type-formatting nil
                           lsp-enable-indentation nil)))

** =age=
#+begin_src emacs-lisp
  (use-package age
    :config
    (setq age-program "rage"
          age-default-recipient nil
          age-default-identity nil)

    ;; age.el's passphrase caching is non-functional (the callback that would
    ;; use it is dead code). Rage prompts via pinentry, bypassing Emacs. We fix
    ;; this by: prompting in the minibuffer, caching per-file, and providing a
    ;; custom pinentry wrapper that returns the cached passphrase via env var.

    (defvar age--passphrase-cache (make-hash-table :test 'equal))

    (defvar age--pinentry-dir
      (let* ((real-pinentry (or (executable-find "pinentry-qt")
                                (executable-find "pinentry-gnome3")
                                (executable-find "pinentry-gtk-2")
                                (executable-find "pinentry")))
             (dir (make-temp-file "emacs-age-pinentry-" t))
             (script (expand-file-name "pinentry" dir)))
        (with-temp-file script
          (insert
           "#!/bin/sh\n"
           "if [ -n \"$EMACS_AGE_PASSPHRASE\" ]; then\n"
           "  echo 'OK'\n"
           "  while IFS= read -r cmd; do\n"
           "    case \"${cmd%% *}\" in\n"
           "      GETPIN) printf 'D %s\\n' \"$EMACS_AGE_PASSPHRASE\"; echo 'OK' ;;\n"
           "      BYE) echo 'OK'; exit 0 ;;\n"
           "      *) echo 'OK' ;;\n"
           "    esac\n"
           "  done\n"
           (if real-pinentry
               (concat "else\n  exec " real-pinentry " \"$@\"\n")
             "")
           "fi\n"))
        (set-file-modes script #o700)
        (dolist (name '("pinentry-qt" "pinentry-gtk-2" "pinentry-gnome3"
                        "pinentry-curses" "pinentry-tty"))
          (make-symbolic-link "pinentry" (expand-file-name name dir) t))
        dir))

    (define-advice age--start (:around (orig-fn context args) use-cached-passphrase)
      "Use cached passphrase via custom pinentry for age operations."
      (if (not (or (age-context-passphrase context) (not age-default-recipient)))
          (funcall orig-fn context args)
        (let* ((file (expand-file-name (or buffer-file-name "")))
               (cached (gethash file age--passphrase-cache))
               (pass (or cached
                         (let ((p (read-passwd
                                   (format "Age passphrase for %s: "
                                           (file-name-nondirectory file))
                                   (eq (age-context-operation context) 'encrypt))))
                           (puthash file (copy-sequence p) age--passphrase-cache)
                           p)))
               (process-environment
                (append (list (concat "EMACS_AGE_PASSPHRASE=" pass)
                              (concat "PATH=" age--pinentry-dir ":" (getenv "PATH")))
                        process-environment)))
          (funcall orig-fn context args))))

    (define-advice age-decrypt-file (:around (orig-fn context cipher plain) clear-cache-on-error)
      "Clear passphrase cache on decryption failure (e.g. wrong passphrase)."
      (condition-case err
          (funcall orig-fn context cipher plain)
        (error
         (remhash (expand-file-name (or buffer-file-name "")) age--passphrase-cache)
         (signal (car err) (cdr err)))))

    (define-advice age-encrypt-string (:around (orig-fn context plain recipients) refuse-empty)
      "Refuse to return empty encrypted output — age.el can miss errors."
      (let ((result (funcall orig-fn context plain recipients)))
        (when (or (null result) (string-empty-p result))
          (error "Age encryption produced empty output — refusing to write"))
        result))

    (defun my/age-ensure-handler ()
      "restore the age handler before saving an age buffer.
without it, saves silently write plaintext. reload-config previously removed
it by repeating init.el setup without rerunning the restoration hook.
keep this before-save-hook guard even with the init.el fix."
      (unless (rassq 'age-file-handler file-name-handler-alist)
        (age-file-enable)
        (message "age: file-name handler was MISSING; re-enabled it before saving %s"
                 (buffer-name))))

    ;; Atomic saves: write to temp file then rename, so a crash mid-write
    ;; can't truncate/destroy the existing encrypted file.
    (add-hook 'find-file-hook
              (lambda ()
                (when (and buffer-file-name
                           (string-match-p "\\.age\\'" buffer-file-name))
                  (setq-local file-precious-flag t)
                  ;; restore the encryption handler before saving.
                  (add-hook 'before-save-hook #'my/age-ensure-handler nil t)
                  ;; Re-sync the recorded modtime after each save. With
                  ;; `file-precious-flag', `age-file-write-region' records the
                  ;; visited modtime BEFORE `basic-save-buffer' renames the temp
                  ;; file into place, so Emacs' recorded modtime lags one save
                  ;; behind the real file -- `verify-visited-file-modtime'
                  ;; returns nil and the modeline shows the file as "changed on
                  ;; disk" right after saving. `after-save-hook' runs after the
                  ;; rename, so re-recording the modtime there fixes the desync.
                  (add-hook 'after-save-hook #'set-visited-file-modtime nil t))))

    (age-encryption-mode))
#+end_src

      "Main text of HTML by eww's readability heuristic, or nil."
      (require 'eww)
      (when-let* ((dom (with-temp-buffer
                         (insert (decode-coding-string html 'undecided))

                  (body (car (dom-by-tag (eww-readable-dom dom) 'body))))
        (dom-texts body " ")))

    (defun my/w3m--locate-text (text)
      "Return (BEG . END) of TEXT's words in the rendered page, or nil.
  Match six-word anchors at each end, skipping up to four boundary words
  to tolerate rendering differences such as image alt text."
      (let* ((words (vconcat (split-string text "[^[:alnum:]]+" t)))
             (n (length words))
             (run 6)
             (case-fold-search nil)
             beg end)
        (cl-flet ((anchor (i) (mapconcat #'regexp-quote (seq-subseq words i (+ i run))
                                         "[^[:alnum:]]+")))
          (when (>= n (+ run 5))
            (save-excursion
              (cl-loop for i below 5 until beg do
                       (goto-char (point-min))
                       (when (re-search-forward (anchor i) nil t)
                         (setq beg (match-beginning 0))))
              (cl-loop for i below 5 until end do
                       (goto-char (point-max))
                       (when (re-search-backward (anchor (- n run i)) nil t)
                         (setq end (match-end 0)))))
            (and beg end (< beg end) (cons beg end))))))

    (defun my/w3m--index-words (beg end)
      "Index the words between BEG and END into `my/w3m-read-table'.
  Cache cumulative line counts so progress updates count only a partial line."
      (save-excursion
        (goto-char beg)
        (let ((starts (list beg)) (counts (list 0)) (n 0))
          (while (< (point) end)
            (let ((from (point)))
              (goto-char (min end (line-beginning-position 2)))
              (setq n (+ n (count-words from (point))))
              (push (point) starts)
              (push n counts)))
          (setq my/w3m-read-table (cons (vconcat (nreverse starts))
                                        (vconcat (nreverse counts)))))))

      "Find the main text of URL's rendered page and index it for reading progress."
      (unless (equal my/w3m-read-key (cons url (buffer-chars-modified-tick)))
        (my/w3m--index-page url)))

    (defun my/w3m--index-page (url)
      (setq my/w3m-read-table nil
            my/w3m-read-key (cons url (buffer-chars-modified-tick)))
      (let ((gen (cl-incf my/w3m-read-gen))
            (buf (current-buffer))
            (html (my/w3m--cached-html url))
            (rdrview (executable-find "rdrview")))
        (cl-flet ((finish (text)
                    (when (and (buffer-live-p buf)
                               (= gen (buffer-local-value 'my/w3m-read-gen buf)))
                      (with-current-buffer buf
                        (save-restriction
                          (widen)
                          (let ((region (or (and text (my/w3m--locate-text text))
                                            (cons (point-min) (point-max)))))
                            (my/w3m--index-words (car region) (cdr region))))
                        (force-mode-line-update t)))))
          (cond
           ((not html) (finish nil))
           (rdrview
            (let* ((default-directory temporary-file-directory)
                   (out (generate-new-buffer " *rdrview*"))
                   (proc (make-process
                          :name "rdrview" :buffer out :noquery t
                          :connection-type 'pipe

                                                 (dom-texts (libxml-parse-html-region
                                                             (point-min) (point-max))
                                                            " ")))))
                                (kill-buffer out)
                                (finish (or text (my/w3m--eww-text html)))))))))
              (process-send-string proc html)
              (process-send-eof proc)))

      "Reading progress \"READ/TOTALW PCT%\" for a w3m page, or nil elsewhere."
      (when-let* ((table my/w3m-read-table)
                  (win (if (eq (window-buffer) (current-buffer))
                           (selected-window) ; the window whose tab line is drawn
                         (get-buffer-window))))
        (let* ((starts (car table))
               (counts (cdr table))
               (last (1- (length starts)))
               (total (aref counts last))
               (pos (max (aref starts 0)

                              (if (eq my/w3m-read-position 'point)
                                  (window-point win)
                                (window-end win)))))
               (lo 0)
               (hi last))

          (while (< lo hi)
            (let ((mid (/ (+ lo hi 1) 2)))
              (if (<= (aref starts mid) pos) (setq lo mid) (setq hi (1- mid)))))
          (let ((read (+ (aref counts lo) (count-words (aref starts lo) pos))))
            (format "%d/%dW %d%%" read total
                    (if (zerop total) 100 (/ (* 100 read) total)))))))

                (or default "")

    (defun my/agent-shell-send-command-with-context (&optional arg)
      "Read a command and send it to the current project's agent-shell, tagged
    with a `file:line' reference for context (like claude-code's
    `send-command-with-context'). Uses the active region's line range if set,
    else the current line. With prefix ARG, focus the shell after sending."
      (interactive "P")
      (require 'agent-shell)
      (let* ((shell-buffer (agent-shell--shell-buffer))
             (file (buffer-file-name))
             (cwd (and shell-buffer
                       (with-current-buffer shell-buffer (agent-shell-cwd))))
             ;; built before the minibuffer read, so the region is still active
             (ref (when file
                    (let ((path (if (and cwd (file-in-directory-p file cwd))
                                    (file-relative-name file cwd)
                                  file)))
                      (if (use-region-p)
                          (format "%s:%d-%d" path
                                  (line-number-at-pos (region-beginning) t)
                                  (line-number-at-pos (region-end) t))
                        (format "%s:%d" path (line-number-at-pos (point) t))))))
             (cmd (read-string "Agent command: " nil 'my/agent-shell-command-history))
             (text (if ref (format "%s\n\n%s" cmd ref) cmd))
             (busy (with-current-buffer shell-buffer
                     (ignore-errors (shell-maker-busy)))))
        ;; submit only if idle; if busy, leave it in the input to send with RET
        (agent-shell-insert :text text :submit (not busy)
                            :shell-buffer shell-buffer :no-focus (not arg))
        (when busy
          (message "Agent busy -- left in input, submit (RET) when ready"))))

      "Account limit percentages, e.g. \"25@14:29 4@Sun 0\", or nil without data."
      (let* ((limits (alist-get 'limits my/claude-usage))
             (weekly (seq-find (lambda (l) (equal (alist-get 'kind l) "weekly_all")) limits))
             parts help)
        (seq-doseq (l limits)
          (let* ((kind (alist-get 'kind l))
                 (pct (or (alist-get 'percent l) 0))
                 (reset (ignore-errors (encode-time (iso8601-parse (alist-get 'resets_at l)))))
                 (model (map-nested-elt l '(scope model display_name)))
                 (label (pcase kind
                          ("session" "5h")
                          ("weekly_all" "wk")
                          ("weekly_scoped" (and model (substring model 0 1)))))

                 (same-reset (and (equal kind "weekly_scoped") weekly reset
                                  (when-let* ((w (ignore-errors
                                                   (encode-time (iso8601-parse
                                                                 (alist-get 'resets_at weekly))))))
                                    (< (abs (float-time (time-subtract reset w))) 3600)))))

                                   (_ (concat model " weekly")))
                            pct (if reset (format-time-string "%a %H:%M" reset) "?"))

      "Compact agent-shell mode-line segment, without the busy spinner."
      (let* ((state (agent-shell--state))
             (model-id (agent-shell--current-model-id state))
             (model (or (map-elt (seq-find (lambda (m) (equal (map-elt m :model-id) model-id))
                                           (agent-shell--get-available-models state))
                                 :name)
                        model-id))

             (mode (agent-shell--resolve-session-mode-name
                    (agent-shell--current-mode-id state)
                    (agent-shell--get-available-modes state)))
             (usage (map-elt state :usage))
             (used (map-elt usage :context-used))
             (size (map-elt usage :context-size))
             (settings
              (delq nil
                    (list (when model
                            (my/agent-shell--part
                             ;; "Opus 5.5" -> "Opus5.5", "Default (recommended)" -> "Default"
                             (replace-regexp-in-string
                              "\\([[:alpha:]]\\) \\([0-9]\\)" "\\1\\2"
                              (replace-regexp-in-string " *(.*)" "" model))
                             'agent-shell-model (concat "Model: " model)
                             #'agent-shell-set-session-model))
                          (when effort
                            (my/agent-shell--part
                             (or (cdr (assoc effort my/agent-shell-effort-abbrevs)) (downcase effort))
                             'agent-shell-thought-level (concat "Effort: " effort)
                             #'agent-shell-set-session-thought-level))
                          (when mode
                            (my/agent-shell--part
                             (or (cdr (assoc mode my/agent-shell-mode-abbrevs)) (downcase mode))
                             'agent-shell-session-mode (concat "Mode: " mode)
                             #'agent-shell-set-session-mode))))))
        (string-join
         (delq nil
               (list ""
                     (when agent-shell-command-prefix
                       (propertize "⧉" 'face 'agent-shell-container-indicator
                                   'help-echo "Running in container"))
                     (when settings (string-join settings "·"))
                     (when (and used size (> size 0))
                       (propertize (format "%s/%s" (agent-shell--format-number-compact used)
                                           (agent-shell--format-number-compact size))
                                   'face (my/agent-shell--pct-face (/ (* 100.0 used) size))
                                   'help-echo (agent-shell--format-usage usage)))
                     (when (my/agent-shell--claude-p) (my/claude-usage--string))))
         " ")))

    ;; Use agent-shell's comint truncation mechanism to retain only the tail;
    ;; full transcripts remain in .agent-shell/transcripts/.
    (defun my/agent-shell-trim (&optional keep)
      "Delete all but the last KEEP lines of this agent-shell buffer.
KEEP defaults to 2000; a numeric prefix argument sets it, and a plain
\\[universal-argument] prompts for it."
      (interactive
       (list (cond ((integerp current-prefix-arg) current-prefix-arg)
                   (current-prefix-arg (read-number "Keep lines: " 2000))
                   (t 2000))))
      (unless (derived-mode-p 'agent-shell-mode)
        (user-error "Not in an agent-shell buffer"))
      (when (shell-maker-busy)
        (user-error "Agent is busy; trim once the turn finishes"))
      (let ((before (count-lines (point-min) (point-max)))
            (comint-buffer-maximum-size (or keep 2000))
            ;; Avoid retaining deleted transcript text in undo history.
            (buffer-undo-list t))

** GC
#+begin_src emacs-lisp
  ;; we need to handle multi-frame emacs sessions, where switching frames from
  ;; emacs -> emacs is NOT a valid time to gc. however, when switching in this
  ;; way, there is a small period of time inbetween receiving notice of
  ;; defocusing the first window, and focusing the second window, where no
  ;; windows are focused.  thus, after a focus change we must have a slight
  ;; delay before checking if any emacs frames are focused, then gc only if none
  ;; are.
  (defvar my/gc-reason nil
    "Current collection trigger; nil denotes the allocation threshold.")

  (defun gc-unless-focused ()
    "Will call garbage-collect if there are no focused emacs frames"
    (interactive)
    (unless (cl-some (lambda (frame) (frame-focus-state frame)) (frame-list))
      (let ((my/gc-reason "unfocused"))
        (garbage-collect))))

  ;; Report GC duration and trigger to identify collection pauses.
  ;; Log only to *Messages* during minibuffer input and idle or unfocused
  ;; collections to preserve prompts and avoid background echo-area updates.
  (defvar my/gc-last-elapsed gc-elapsed
    "`gc-elapsed' after the previous collection.")

  (defun my/gc-report ()
    "Report the last collection's duration and trigger."
    (let ((inhibit-message (or (member my/gc-reason '("idle" "unfocused"))
                               (and (active-minibuffer-window) t))))
      (message "GC (%s): %.3fs" (or my/gc-reason "threshold")
               (- gc-elapsed my/gc-last-elapsed)))
    (setq my/gc-last-elapsed gc-elapsed))
  (add-hook 'post-gc-hook #'my/gc-report)

  (add-function :after after-focus-change-function
                (lambda () (run-with-timer 0.1 nil #'gc-unless-focused)))

  ;; Raise the GC threshold during commands; collect and lower it on idle.
  ;; Adapt the idle delay to 10x the previous idle collection's duration.
  (use-package gcmh
    :config
    (setq gcmh-idle-delay 'auto
          gcmh-auto-idle-delay-factor 10
          ;; On this 32 GB system, heap scans take ~200-500 ms per collection.
          ;; Use 256 MiB (4x Doom's default) to reduce collections during input.
          gcmh-high-cons-threshold (* 256 1024 1024)
          ;; Match init.el's post-startup threshold. GCMH's 800 kB default
          ;; causes frequent ~250 ms collections during agent-shell output
          ;; without input; 64 MiB halves the previous 32 MiB collection rate.
          gcmh-low-cons-threshold (* 64 1024 1024))
    (define-advice gcmh-idle-garbage-collect (:around (orig) my/gc-reason)
      (let ((my/gc-reason "idle"))
        (funcall orig)))
    (gcmh-mode 1))
#+end_src
